% TikZ 图：圆锥摆的受力与几何关系
% Day 11：圆周运动
\begin{tikzpicture}[scale=0.88,>=Latex]
  \coordinate (O) at (0,0);
  \coordinate (P) at (1.45,-2.65);
  \coordinate (C) at (0,-2.65);

  \draw[thick] (-1.1,0) -- (1.1,0);
  \filldraw[black] (O) circle (2pt);
  \draw[dashed] (O) -- (C);
  \draw[thick] (O) -- (P) node[midway,right=2pt] {$l$};
  \draw[dashed] (C) ellipse (1.45 and 0.32);
  \draw[densely dashed] (C) -- (0,-3.32);
  \draw[densely dashed] (P) -- (1.45,-3.32);
  \draw[<->,thick] (0,-3.32) -- (1.45,-3.32)
    node[midway,below=4pt,fill=white,inner sep=1pt] {$r=l\sin\theta$};

  \filldraw[red] (P) circle (3pt);
  \node[right=4pt] at (P) {$m$};
  % 实际受力只有拉力和重力；分量关系由右侧方程说明。
  \draw[->,thick,blue] (P) -- ($(P)!0.48!(O)$)
    node[pos=0.52,left=5pt,fill=white,inner sep=1pt] {$F_T$};
  \draw[->,thick,red] (P) -- ++(0,-0.92)
    node[pos=0.82,right=5pt,fill=white,inner sep=1pt] {$mg$};

  \draw (0,-0.45) arc[start angle=-90,end angle=-61.3,radius=0.45];
  \node at (0.28,-0.70) {$\theta$};
  \node[align=center] at (3.35,-1.15)
    {$F_T\cos\theta=mg$\\[3pt]
     $F_T\sin\theta=m\omega^2r$};
\end{tikzpicture}
