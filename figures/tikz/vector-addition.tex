% TikZ 图：矢量加法的平行四边形法则
% Day 01：初高中物理的区别与学习方法
% 使用：\input{figures/tikz/vector-addition.tex}
\begin{tikzpicture}[scale=1.4,>=Latex]
  \coordinate (O) at (0,0);
  \coordinate (A) at (3.2,0);
  \coordinate (B) at (0,4.2);
  \coordinate (C) at (3.2,4.2);

  % 两个分矢量从同一起点出发
  \draw[->,thick] (O) -- (A) node[midway,below] {$3\ \mathrm{m}$};
  \draw[->,thick] (O) -- (B) node[midway,left] {$4\ \mathrm{m}$};

  % 平行四边形辅助线
  \draw[dashed] (A) -- (C);
  \draw[dashed] (B) -- (C);

  % 合矢量
  \draw[->,ultra thick,red] (O) -- (C) node[midway,above left] {$5\ \mathrm{m}$};

  \node[below left] at (O) {起点};
  \node[below] at (A) {东};
  \node[left] at (B) {北};
  \node[above right] at (C) {合矢量};
\end{tikzpicture}
