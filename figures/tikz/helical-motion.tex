% TikZ 图：斜入磁场的螺旋运动
% Day 22：洛伦兹力
\begin{tikzpicture}[scale=0.9,>=Latex]
  \foreach \x in {-2.2,-1.2,-0.2,0.8,1.8,2.8} {
    \draw[->,blue] (\x,-1.4) -- (\x,1.35);
  }
  \node[blue] at (3.2,1.1) {$B$};
  \draw[->,black,thick] (-2.5,0) -- (3.2,0) node[right] {沿磁场方向};
  \draw[red,thick,domain=-2.2:2.4,samples=120,->]
    plot ({\x},{0.55*sin(360*\x)});
  \draw[->,orange,thick] (-2.2,0) -- (-1.35,0.75) node[above] {$v$};
  \draw[->,orange] (-2.2,0) -- (-1.35,0) node[midway,below] {$v_\parallel$};
  \draw[->,orange] (-1.35,0) -- (-1.35,0.75) node[midway,right] {$v_\perp$};
  \node at (0,-1.85) {圆周运动 + 匀速直线运动 = 螺旋线};
\end{tikzpicture}
