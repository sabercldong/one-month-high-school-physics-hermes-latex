% TikZ 图：条形磁体磁感线
% Day 21：磁场与安培力
\begin{tikzpicture}[scale=0.92,>=Latex,line cap=round,line join=round]
  \path[use as bounding box] (-4.05,-2.20) rectangle (4.05,2.25);

  % 外部磁感线：从 N 极出发，在磁体外部绕到 S 极进入。
  \draw[blue,thick,->] (-2.25,0.32)
    .. controls (-3.70,2.05) and (3.70,2.05) .. (2.25,0.32);
  \draw[blue,thick,->] (-2.25,0.22)
    .. controls (-3.30,1.52) and (3.30,1.52) .. (2.25,0.22);
  \draw[blue,thick,->] (-2.25,0.11)
    .. controls (-2.92,1.02) and (2.92,1.02) .. (2.25,0.11);
  \draw[blue,thick,->] (-2.25,-0.11)
    .. controls (-2.92,-1.02) and (2.92,-1.02) .. (2.25,-0.11);
  \draw[blue,thick,->] (-2.25,-0.22)
    .. controls (-3.30,-1.52) and (3.30,-1.52) .. (2.25,-0.22);
  \draw[blue,thick,->] (-2.25,-0.32)
    .. controls (-3.70,-2.05) and (3.70,-2.05) .. (2.25,-0.32);

  % 条形磁体。
  \draw[thick,fill=red!10] (-2.20,-0.38) rectangle (0,0.38);
  \draw[thick,fill=blue!10] (0,-0.38) rectangle (2.20,0.38);
  \draw[thick] (0,-0.38) -- (0,0.38);
  \node[font=\large] at (-1.10,0) {$N$};
  \node[font=\large] at (1.10,0) {$S$};

  % 磁体内部磁感线由 S 极回到 N 极，与外部磁感线共同构成闭合曲线。
  \draw[blue!65,dashed,thick,->] (1.78,0.18) -- (-1.78,0.18);
  \draw[blue!65,dashed,thick,->] (1.78,-0.18) -- (-1.78,-0.18);

  \node[blue,font=\small,fill=white,inner sep=1pt] at (0,1.78)
    {外部：$N\to S$};
  \node[blue!70,font=\small,fill=white,inner sep=1pt] at (0,-1.78)
    {内部：$S\to N$，整体闭合};
\end{tikzpicture}
