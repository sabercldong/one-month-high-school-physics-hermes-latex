% Day 05：悬挂法找二维平面薄板的重心
\begin{tikzpicture}[scale=1.0,>=Latex,line cap=round,line join=round,
  every node/.style={font=\small}]

  % 从孔 A 悬挂：旋转后 A、重心 C 位于同一条竖直线上。
  \begin{scope}[shift={(-2.65,0.48)},rotate=-61.19]
    \coordinate (Cone) at (0,0);
    \coordinate (Aone) at (-1.00,0.55);
    \filldraw[fill=gray!12,draw=black!75,thick]
      (-1.10,-0.65) -- (0.90,-0.55) -- (1.20,0.40) --
      (0.20,1.00) -- (-1.00,0.55) -- cycle;
    \fill (Aone) circle (1.5pt);
  \end{scope}
  \draw[thick] ($(Aone)+(-0.24,0.15)$) -- ($(Aone)+(0.24,0.15)$);
  \draw[dashed,gray!70] (Aone) -- ($(Aone)+(0,-2.05)$);
  \draw[->,blue!75!black,thick] (Cone) -- ++(0,-0.70)
    node[right=1pt] {$G$};
  \node[above left=1pt] at (Aone) {$A$};
  \node[font=\footnotesize] at (-2.65,-1.20) {从孔 $A$ 悬挂};

  % 从孔 B 悬挂：换孔后薄板转到另一稳定姿态。
  \begin{scope}[shift={(2.65,0.48)},rotate=71.565]
    \coordinate (Ctwo) at (0,0);
    \coordinate (Btwo) at (1.20,0.40);
    \filldraw[fill=gray!12,draw=black!75,thick]
      (-1.10,-0.65) -- (0.90,-0.55) -- (1.20,0.40) --
      (0.20,1.00) -- (-1.00,0.55) -- cycle;
    \fill (Btwo) circle (1.5pt);
  \end{scope}
  \draw[thick] ($(Btwo)+(-0.24,0.15)$) -- ($(Btwo)+(0.24,0.15)$);
  \draw[dashed,gray!70] (Btwo) -- ($(Btwo)+(0,-2.05)$);
  \draw[->,blue!75!black,thick] (Ctwo) -- ++(0,-0.70)
    node[right=1pt] {$G$};
  \node[above right=1pt] at (Btwo) {$B$};
  \node[font=\footnotesize] at (2.65,-1.20) {从孔 $B$ 悬挂};

  \node[black!78,font=\footnotesize] at (0,-1.55)
    {每次稳定后，重心都在悬挂点的正下方};

  % 将两次悬挂所得的铅垂线转画到同一薄板上。
  \begin{scope}[shift={(0,-3.20)}]
    \coordinate (C) at (0,0);
    \coordinate (A) at (-1.00,0.55);
    \coordinate (B) at (1.20,0.40);
    \filldraw[fill=gray!12,draw=black!75,thick]
      (-1.10,-0.65) -- (0.90,-0.55) -- (1.20,0.40) --
      (0.20,1.00) -- (-1.00,0.55) -- cycle;
    \fill (A) circle (1.3pt) node[above left=1pt] {$A$};
    \fill (B) circle (1.3pt) node[above right=1pt] {$B$};
    \draw[dashed,red!75!black,thick]
      ($(A)!-0.12!(C)$) -- ($(A)!1.48!(C)$);
    \draw[dashed,blue!70!black,thick]
      ($(B)!-0.10!(C)$) -- ($(B)!1.42!(C)$);
    \filldraw[fill=red!85!black,draw=red!85!black] (C) circle (2.2pt);
    \draw[red!75!black] (C) -- (1.48,0)
      node[right=2pt] {重心};
    \node[red!75!black,font=\footnotesize,anchor=east] at (-1.22,-0.22)
      {第 1 次铅垂线};
    \node[blue!70!black,font=\footnotesize,anchor=west] at (1.25,-0.38)
      {第 2 次铅垂线};
  \end{scope}
  \node[align=center,font=\footnotesize] at (0,-4.28)
    {把两次铅垂线转画到同一薄板上，交点即为重心};
\end{tikzpicture}
