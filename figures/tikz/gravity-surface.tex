% TikZ 图：地球表面重力与万有引力

\begin{tikzpicture}[scale=0.7,>=Latex]
  % 地球
  \shadedraw[ball color=blue!20,draw=blue] (0,0) circle (2);

  \filldraw[red] (45:2) circle (3pt);

  % 自转轴
