% TikZ 图：斜面上牛顿第二定律（下滑时）
% Day 06：牛顿运动定律
\begin{tikzpicture}[scale=0.92,>=Latex]
  % 斜面：物体放在上表面，右下方为下滑方向。
  \draw[thick] (0,0) -- (5,0) -- (0,2) -- cycle;
  \draw[thick] (0,2) -- (5,0);
  \draw (4.25,0) arc (180:158.2:0.75);
  \node at (4.1,0.25) {$\theta$};

  \coordinate (P) at (2.35,1.06);
  \coordinate (C) at ($(P)+(68.2:0.42)$);

  \begin{scope}[shift={(P)},rotate=-21.8]
    \draw[thick] (-0.6,0) rectangle (0.6,0.85);
    \node at (0,0.43) {$m$};
  \end{scope}

  % 重力分力，用虚线区分，避免误认为额外真实力。
  \draw[->,thick,red,dashed] (C) -- ++(-21.8:1.25) node[below right] {$G\sin\theta$};
  \draw[->,thick,blue,dashed] (C) -- ++(-111.8:0.75) node[pos=0.75,left] {$G\cos\theta$};

  % 真实受力：重力、支持力、摩擦力。
  \draw[->,thick,blue] (C) -- ++(0,-1.15) node[right] {$G$};
  \draw[->,thick,blue] (C) -- ++(68.2:1.1) node[above] {$N$};
  \draw[->,thick,red] (C) -- ++(158.2:1.05) node[above] {$f$};

  % 加速度方向。
  \draw[->,ultra thick,red] ($(C)+(68.2:0.62)+(158.2:0.2)$) -- ++(-21.8:1.35) node[above right] {$a$};
\end{tikzpicture}
