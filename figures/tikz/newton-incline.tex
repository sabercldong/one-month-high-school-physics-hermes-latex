% TikZ 图：斜面上牛顿第二定律（下滑时）
% Day 06：牛顿运动定律
% 左高右低斜面，受真实力（G/N/f）+ 重力分力（虚线）
\begin{tikzpicture}[scale=0.9,>=Latex]
  % 斜面（左高右低）
  \draw[thick] (0,2.5) -- (5,0.5);
  \draw[thick] (0,0.5) -- (5,0.5);
  % 角度标注在左下角
  \draw (0.8,0.5) arc (0:-21.8:0.8);
  \node at (1.1,0.35) {$\theta$};
  % 物体
  \draw[thick,rotate around={-21.8:(2.5,1.3)}] (1.9,0.8) rectangle (3.1,1.8);
  \node[rotate=-21.8] at (2.5,1.3) {$m$};
  
  % === 真实受力（箭头粗，实线）===
  % 重力 G（竖直向下）
  \draw[->,thick,blue] (2.5,0.8) -- (2.5,-0.3) node[below] {$G$};
  % 支持力 N（垂直斜面向上）
  \draw[->,thick,blue,rotate around={-21.8:(2.5,1.3)}] (2.5,1.8) -- (2.5,2.7) node[above] {$N$};
  % 摩擦力 f（下滑时沿斜面向上）
  \draw[->,thick,red,rotate around={-21.8:(2.5,1.3)}] (1.9,1.3) -- (1.1,1.3) node[above] {$f$};
  % 加速度 a
  \draw[->,ultra thick,red,rotate around={-21.8:(2.5,1.3)}] (2.5,2.0) -- (4,2.0) node[above] {$a$};
  
  % === 重力分力（虚线，较细）===
  % G sinθ 沿斜面向下（虚线，橙色）
  \draw[->,thick,dashed,orange,rotate around={-21.8:(2.5,1.3)}] (2.5,0.8) -- (4,0.8) node[right] {$G\sin\theta$};
  % G cosθ 垂直压向斜面（虚线，橙色）
  \draw[->,thick,dashed,orange,rotate around={-21.8:(2.5,1.3)}] (2.5,0.8) -- (2.5,-0.3) node[below] {$G\cos\theta$};
  
  % 条件说明
  \node[below,font=\footnotesize] at (2.5,-0.6) {（下滑时 $f$ 沿斜面向上，虚线为重力分力）};
\end{tikzpicture}
