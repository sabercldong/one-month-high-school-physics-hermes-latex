% TikZ 图：先加速后偏转模型
\begin{tikzpicture}[scale=0.94,>=Latex,font=\small]
  % 加速区：两电极均留有小孔
  \foreach \x in {-3.2,-1.6} {
    \draw[thick] (\x,1.0) -- (\x,0.17);
    \draw[thick] (\x,-0.17) -- (\x,-1.0);
  }
  \node[above] at (-3.2,1.04) {$+$};
  \node[above] at (-1.6,1.04) {$-$};
  \foreach \y in {-0.48,0.48} {
    \draw[->,blue] (-3.0,\y) -- (-1.8,\y);
  }
  \draw[<->] (-3.2,1.32) -- (-1.6,1.32);
  \node[above,blue,fill=white,inner sep=1pt] at (-2.4,1.32) {$U_1$};
  % The particle starts just inside the accelerating gap, where the
  % idealized electric field is present.
  \filldraw[red] (-2.96,0) circle (2pt);
  \node[below left,red] at (-2.96,0) {$+q$};
  \draw[red,thick,dashed] (-2.82,0) -- (-1.62,0);
  \draw[->,red,thick] (-1.40,0) -- (-0.48,0);
  \node[red,above] at (-0.94,0) {$\vect v_0$};

  % 偏转区
  \draw[thick] (-0.2,1.2) -- (3.0,1.2);
  \draw[thick] (-0.2,-1.2) -- (3.0,-1.2);
  \node[above] at (-0.35,1.2) {$+$};
  \node[below] at (-0.35,-1.2) {$-$};
  \foreach \x in {0.35,1.05,1.75,2.45} {
    \draw[->,blue] (\x,1.0) -- (\x,-1.0);
  }
  \draw[<->] (-0.2,1.55) -- (3.0,1.55);
  \node[above,fill=white,inner sep=1pt] at (1.4,1.55) {$L$};
  \draw[<->] (3.30,1.2) -- (3.30,-1.2);
  \node[right] at (3.30,0.48) {$d$};
  \node[blue] at (0.18,-1.48) {$U_2$};

  % 抛物线只画到出口；出口速度沿切线
  \draw[red,thick,dashed,domain=-0.2:3.0,samples=80]
    plot (\x,{-0.10*(\x+0.2)*(\x+0.2)});
  \coordinate (P) at (3.0,-1.024);
  \filldraw[red] (P) circle (2pt);
  \draw[->,red,thick] (P) -- (3.62,-1.42);

  \draw[dashed,gray] (3.0,0) -- (P);
  \draw[<->] (3.83,0) -- (3.83,-1.024);
  \node[right] at (3.83,-0.52) {$|y|$};
  \node at (-2.4,-1.42) {加速区};
  \node at (1.75,-1.72) {偏转区};
\end{tikzpicture}
