% TikZ 图：电磁波中 E、B 与传播方向两两垂直
% Day 24：交变电流、变压器与电磁应用
\begin{tikzpicture}[scale=0.90,>=Latex,line cap=round,line join=round]
  % Three projected axes.
  \draw[->,thick] (0,0) -- (7.35,0) node[right] {传播方向};
  \draw[->,thick,blue] (0,0) -- (0,1.72) node[above] {$\vec E$};
  \draw[->,thick,red!75!black] (0,0) -- (-0.82,-0.82)
    node[below left] {$\vec B$};

  % E wave in the x-y plane.
  \draw[blue,very thick,samples=160,domain=0:6.75]
    plot(\x,{1.02*sin(180*\x/1.50)});

  % B wave along a projected z direction: displacement basis (-0.48,-0.48).
  \draw[red!75!black,very thick,samples=160,domain=0:6.75]
    plot({\x-0.48*sin(180*\x/1.50)},{-0.48*sin(180*\x/1.50)});

  \foreach \x in {0.75,2.25,3.75,5.25,6.75}{
    \draw[blue!65,->] (\x,0) -- (\x,{1.02*sin(180*\x/1.50)});
    \draw[red!60!black,->] (\x,0) --
      ({\x-0.48*sin(180*\x/1.50)},{-0.48*sin(180*\x/1.50)});
  }

  \node[font=\small,align=center] at (3.55,-1.62)
    {$\vec E\perp\vec B$，且二者均垂直于传播方向；\\
     电场和磁场同相振荡并向前传播};
\end{tikzpicture}
