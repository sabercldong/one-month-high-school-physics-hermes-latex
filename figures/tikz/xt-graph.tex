% TikZ 图：x-t 图像示例（匀速直线）
% Day 04：运动图像与追及相遇
\begin{tikzpicture}[scale=1,>=Latex]
  \draw[->] (-0.3,0) -- (5.5,0) node[right] {$t$};
  \draw[->] (0,-0.3) -- (0,4.5) node[above] {$x$};
  \node[below left] at (0,0) {$O$};
  % 匀速直线
  \draw[very thick] (0,0) -- (5,4);
  \node[above left] at (4.55,3.65) {$x=vt$};
  \node[align=center] at (2.45,3.55) {斜率\\[-1pt]$v=\dfrac{\Delta x}{\Delta t}$};
  % 完整直角三角形：避开坐标轴，突出斜率是两点之间的变化率
  \coordinate (A) at (1.4,1.12);
  \coordinate (B) at (3.6,2.88);
  \draw[dashed,thick] (A) -- (3.6,1.12) -- (B);
  \fill (A) circle (1.5pt);
  \fill (B) circle (1.5pt);
  \node[below] at (2.5,1.12) {$\Delta t$};
  \node[right] at (3.6,2.0) {$\Delta x$};
\end{tikzpicture}
