% TikZ 图：x-t 图像示例（匀速直线）
% Day 04：运动图像与追及相遇
\begin{tikzpicture}[scale=1,>=Latex]
  \draw[->] (-0.3,0) -- (5.5,0) node[right] {$t$};
  \draw[->] (0,-0.3) -- (0,4.5) node[above] {$x$};
  \node[below left] at (0,0) {$O$};
  % 匀速直线
  \draw[very thick] (0,0) -- (5,4);
  % 完整直角三角形：水平边 Δt，竖直边 Δx，斜边在直线上
  \draw[dashed,thick] (0,0) -- (3.5,0);   % 水平边 = t
  \draw[dashed,thick] (3.5,0) -- (3.5,2.8); % 竖直边 = x
  % 标注
  \node[below] at (1.75,0) {$\Delta t$};
  \node[right] at (3.5,1.4) {$\Delta x$};
  \node[above right] at (0.5,0.8) {斜率 $= \dfrac{\Delta x}{\Delta t} = v$};
\end{tikzpicture}
