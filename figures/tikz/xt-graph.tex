% TikZ 图：x-t 图像示例（匀速直线）
% Day 04：运动图像与追及相遇
\begin{tikzpicture}[scale=1,>=Latex]
  \draw[->] (-0.3,0) -- (5.3,0) node[right] {$t$};
  \draw[->] (0,-0.3) -- (0,4.3) node[above] {$x$};
  \node[below left] at (0,0) {$O$};

  % 匀速直线
  \draw[very thick] (0,0) -- (4.7,3.55);
  \node[anchor=west] at (0.95,3.45) {斜率：$v=\dfrac{\Delta x}{\Delta t}$};

  % 完整直角三角形：避开坐标轴，突出斜率是两点之间的变化率
  \coordinate (A) at (1.35,1.02);
  \coordinate (B) at (3.35,2.53);
  \coordinate (C) at (3.35,1.02);
  \draw[dashed,thick] (A) -- (C) -- (B);
  \fill (A) circle (1.5pt);
  \fill (B) circle (1.5pt);
  \node[below] at (2.35,1.02) {$\Delta t$};
  \node[right] at (3.35,1.78) {$\Delta x$};
\end{tikzpicture}
