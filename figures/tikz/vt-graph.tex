\begin{tikzpicture}[scale=1,>=Latex]
  % 坐标轴
  \draw[->] (-0.3,0) -- (5.5,0) node[right] {$t$};
  \draw[->] (0,-0.3) -- (0,4.5) node[above] {$v$};
  % 标注原点
  \node[below left] at (0,0) {$O$};
  % 画 v-t 线（黑色粗实线）
  \draw[very thick] (0,1) -- (4,4);
  % 标注 v0
  \draw[->,thick] (0,0) -- (0,1);
  \node[left] at (0,0.5) {$v_0$};
  % 标注某时刻 t
  \draw[dashed] (4,0) -- (4,4);
  \node[below] at (4,0) {$t$};
  % v 标注
  \node[right] at (4.2,2.5) {$v = v_0 + at$};
  % 梯形面积标注
  \fill[black!15] (0,0) -- (0,1) -- (4,4) -- (4,0) -- cycle;
  \node at (2.8,0.8) {$x = \frac{v_0+v}{2}t$};
  % 斜率标注：用与图线平行的虚线
  \draw[dashed] (1.5,2.125) -- (2.5,2.125);
  \draw[dashed] (2.5,2.125) -- (2.5,3.25);
