% TikZ 图：匀变速直线运动的 v-t 图像
% Day 03：加速度
% 使用：\input{figures/tikz/vt-graph.tex}
% 直线从 (0,v0=1) 到 (t=4,v=4)
\begin{tikzpicture}[scale=1.1,>=Latex]
  % 坐标轴
  \draw[->] (-0.3,0) -- (5.5,0) node[right] {$t$};
  \draw[->] (0,-0.3) -- (0,4.8) node[above] {$v$};
  \node[below left] at (0,0) {$O$};
  % 梯形面积先填充，再画边界线，避免遮住图线
  \fill[black!15] (0,0) -- (0,1) -- (4,4) -- (4,0) -- cycle;
  % v-t 直线：从 (0,v0) 到 (t,v)
  \draw[very thick] (0,1) -- (4,4);
  % v0 标注
  \node[left] at (0,1) {$v_0$};
  % t 时刻虚线 + v 标注在顶点
  \draw[dashed] (4,0) -- (4,4);
  \node[below] at (4,0) {$t$};
  \node[right] at (4,4) {$v$};
  \node at (2.45,0.45) {$x=\dfrac{v_0+v}{2}t$};
  % 完整斜率三角形：从 (0,v0) 到 (t,v)
  \draw[dashed,thick] (0,1) -- (4,1);   % 水平边 = t
  \draw[dashed,thick] (4,1) -- (4,4);   % 竖直边 = v - v0
  \node[above] at (2,1) {$\Delta t=t$};
  \node[right] at (4,2.5) {$v - v_0$};
  % 斜边就是 v-t 直线本身（不重复画）
  % 斜率标注放在图像上方空白处，避免压到直线与辅助线
  \node[anchor=west,fill=white,inner sep=2pt] at (0.35,4.45) {斜率 $a = \dfrac{v-v_0}{t}$};
\end{tikzpicture}
