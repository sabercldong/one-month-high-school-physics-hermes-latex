% TikZ 图：竖直圆周运动（内侧轨道/过山车模型）
% Day 11：圆周运动
\begin{tikzpicture}[scale=0.72,>=Latex]
  \draw[thick] (0,0) circle (2.5);
  \filldraw[black] (0,0) circle (2pt);
  \node[left] at (0,0) {$O$};

  % 最高点：重力和支持力都指向圆心
  \filldraw[blue] (0,2.5) circle (3pt);
  \node[above] at (0,2.8) {最高点};
  \draw[->,thick,blue] (0,2.5) -- (1.05,2.5) node[above right] {$v$};
  \draw[->,thick,red] (0,2.5) -- (0,1.32)
    node[pos=0.78,left=2pt,fill=white,inner sep=1pt] {$mg$};
  \draw[->,thick,orange] (0,2.40) -- (0,1.72)
    node[pos=0.25,right=2pt,fill=white,inner sep=1pt] {$N$};

  % 最低点：支持力向上，重力向下
  \filldraw[blue] (0,-2.5) circle (3pt);
  \node[below right] at (0.22,-2.82) {最低点};
  \draw[->,thick,blue] (0,-2.5) -- (1.05,-2.5) node[above right] {$v$};
  \draw[->,thick,red] (0,-2.5) -- (0,-3.34)
    node[pos=0.72,left=2pt,fill=white,inner sep=1pt] {$mg$};
  \draw[->,thick,orange] (0,-2.40) -- (0,-1.46)
    node[pos=0.78,right=2pt,fill=white,inner sep=1pt] {$N$};

  \node[right] at (2.85,1.0) {最高点：$N+mg=m\dfrac{v^2}{r}$};
  \node[right] at (2.85,0.25) {临界：$N=0$};
  \node[right] at (2.85,-0.75) {最低点：$N-mg=m\dfrac{v^2}{r}$};
\end{tikzpicture}
