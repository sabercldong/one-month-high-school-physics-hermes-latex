% TikZ 图：有界磁场圆弧几何
% Day 22：洛伦兹力
\begin{tikzpicture}[scale=0.9,>=Latex]
  \draw[thick,fill=blue!5] (-0.2,-1.8) rectangle (4.2,1.8);
  \foreach \x in {0.3,1.1,1.9,2.7,3.5} {
    \foreach \y in {-1.2,-0.4,0.4,1.2} {
      \draw[gray] (\x-0.05,\y-0.05) -- (\x+0.05,\y+0.05);
      \draw[gray] (\x-0.05,\y+0.05) -- (\x+0.05,\y-0.05);
    }
  }
  \node[blue] at (0.65,1.55) {$B$ 向里};

  \coordinate (A) at (-0.2,-0.9);
  \coordinate (B) at (2.8,1.35);
  \coordinate (O) at (1.35,-0.15);
  \draw[red,very thick,->] (A) .. controls (0.15,0.2) and (1.45,1.4) .. (B);
  \fill[red] (A) circle (2pt) node[left] {入射点};
  \fill[red] (B) circle (2pt) node[below right] {出射点};
  \fill[black] (O) circle (1.8pt) node[below] {$O$};

  \draw[->,red,thick] (-1.2,-0.9) -- (A) node[midway,below] {$v_1$};
  \draw[->,red,thick] (B) -- (3.7,1.35) node[right] {$v_2$};
  \draw[dashed,thick] (O) -- (A) node[midway,below left] {$r$};
  \draw[dashed,thick] (O) -- (B) node[midway,right] {$r$};
  \draw[orange,thick,->] (1.75,0.05) arc (20:72:0.65);
  \node[orange] at (2.05,0.55) {$\alpha$};
  \node[font=\small,align=center] at (2.0,-2.25)
    {半径垂直速度；圆心角 $\alpha$ 决定运动时间 $t=\dfrac{\alpha}{2\pi}T$};
\end{tikzpicture}
