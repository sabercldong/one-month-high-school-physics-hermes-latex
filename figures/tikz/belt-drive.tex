\begin{tikzpicture}[scale=0.8,>=Latex,line cap=round,line join=round]
  % 主动轮
  \draw[thick] (0,0) circle (1.5);
  \filldraw[black] (0,0) circle (2pt);
  \node[below=7pt] at (0,-1.5) {$A$};
  \draw[->,thin] (0,0) -- (-1.06,-1.06) node[midway,above left=-1pt] {$r_A$};
  % 从动轮
  \draw[thick] (4,0) circle (1.0);
  \filldraw[black] (4,0) circle (2pt);
  \node[below=7pt] at (4,-1.0) {$B$};
  \draw[->,thin] (4,0) -- (4.71,-0.71) node[midway,above right=-1pt] {$r_B$};
  % 皮带（上下各一条线）
  \draw[thick] (0,1.5) -- (4,1.0);
  \draw[thick] (0,-1.5) -- (4,-1.0);
  % 角速度标注
  \draw[->,thick,red] (0,0) ++(30:0.6) arc (30:120:0.6);
  \node[red] at (0,0.9) {$\omega_A$};
  \draw[->,thick,blue] (4,0) ++(30:0.5) arc (30:120:0.5);
  \node[blue] at (4,0.7) {$\omega_B$};
  % 公式
  \node[align=center,font=\small] at (2,-2.55)
    {$v_A = v_B$\\$\omega_A r_A = \omega_B r_B$};
