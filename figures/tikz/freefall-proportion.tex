% TikZ 图：自由落体等时间间隔位移示意
% Day 03：加速度
% 使用：\input{figures/tikz/freefall-proportion.tex}
% 第1/2/3秒内位移精确按 1:3:5 比例绘制（5:15:25）
% 段长比例：第1段 0.6, 第2段 1.8 (=0.6×3), 第3段 3.0 (=0.6×5)
\begin{tikzpicture}[scale=0.85,>=Latex]
  % 顶部释放点
  \draw[thick] (-0.5,0) -- (0.5,0);
  \filldraw[black] (0,0) circle (2pt) node[above] {释放点};
  % 第1秒内：长度 0.6 (对应 5m)
  \draw[dashed] (0,0) -- (0,-0.6);
  \filldraw[black] (0,-0.6) circle (2pt);
  \node[right] at (0.2,-0.3) {第$1\ \mathrm{s}$内：$5\ \mathrm{m}$};
  % 第2秒内：长度 1.8 = 0.6×3 (对应 15m)
  \draw[dashed] (0,-0.6) -- (0,-2.4);
  \filldraw[black] (0,-2.4) circle (2pt);
  \node[right] at (0.2,-1.5) {第$2\ \mathrm{s}$内：$15\ \mathrm{m}$};
  % 第3秒内：长度 3.0 = 0.6×5 (对应 25m)
  \draw[dashed] (0,-2.4) -- (0,-5.4);
  \filldraw[black] (0,-5.4) circle (2pt);
  \node[right] at (0.2,-3.9) {第$3\ \mathrm{s}$内：$25\ \mathrm{m}$};
  % g 箭头：移到右侧远处，避免与文字重叠
  \draw[->,thick] (2.5,0) -- (2.5,-5.4) node[midway,right,xshift=3pt] {$g$};
  % 图注
  \draw[dashed,thick] (-0.6,-5.6) -- (0.6,-5.6);
  \node[below] at (0,-5.6) {地面（以 $g=10\ \mathrm{m/s^2}$ 为例，总高 $45\ \mathrm{m}$）};
\end{tikzpicture}
