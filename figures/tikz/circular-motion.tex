\begin{tikzpicture}[scale=0.8,>=Latex]

  % 轨道

  % 圆心
  \filldraw[black] (0,0) circle (2pt);
  \node[below left] at (0,0) {$O$};
  % 半径与角度参考边
  \draw[dashed,gray] (0,0) -- (0:2.15);
