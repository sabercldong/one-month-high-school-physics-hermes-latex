% TikZ 图：微平移法例题
% Day 14：机械波
\begin{tikzpicture}[x=0.95cm,y=0.76cm,>=Latex]
  \draw[->] (-0.35,0) -- (6.7,0) node[right] {$x$};
  \draw[->] (0,-1.25) -- (0,1.35) node[above] {$y$};
  \draw[very thick,red,smooth,domain=0:6.05,samples=120]
    plot (\x,{0.85*sin(120*\x)});
  \draw[dashed,gray,smooth,domain=0.22:6.27,samples=120]
    plot (\x,{0.85*sin(120*(\x-0.22))});

  \coordinate (P) at (1.35,{0.85*sin(120*1.35)});
  \coordinate (Q) at (1.35,{0.85*sin(120*(1.35-0.22))});
  \filldraw[black] (P) circle (2pt) node[below left] {$P$};
  \draw[->,blue,very thick] (P) -- (Q);
  \node[blue,anchor=west] at (1.45,0.82) {$v_P$ 向上};
  \draw[->,thick] (4.9,1.05) -- (6.1,1.05) node[right] {波向右};
  \node[gray,font=\small] at (3.25,-1.05) {虚线表示极短时间后的波形};
\end{tikzpicture}
