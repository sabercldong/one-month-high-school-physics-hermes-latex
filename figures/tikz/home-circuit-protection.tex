% TikZ 图：家庭电路、空气开关、漏电保护与保护接地
% Day 24：交变电流、变压器与电磁应用
\begin{tikzpicture}[scale=0.88,>=Latex,line cap=round,line join=round,
  every node/.style={font=\small}]
  \path[use as bounding box] (-0.55,-2.40) rectangle (10.70,2.35);
  \tikzset{
    live/.style={thick,red!80!black},
    neutral/.style={thick,blue!70!black},
    earth/.style={thick,green!45!black},
    box/.style={draw,rounded corners=3pt,fill=gray!8}
  }

  \node[box,minimum width=1.35cm,minimum height=1.45cm,align=center] (supply) at (0.35,0.55)
    {家庭\\配电端};
  \node[box,fill=orange!10,minimum width=1.45cm,minimum height=0.72cm]
    (breaker) at (2.55,1.15) {空气开关};

  % RCD encloses both live and neutral conductors.
  \draw[rounded corners=4pt,fill=orange!7,thick] (4.15,-0.15) rectangle (6.35,1.78);
  \node[font=\small\bfseries] at (5.25,1.52) {漏电保护器};

  \node[box,minimum width=1.45cm,minimum height=1.25cm] (load) at (8.05,0.55)
    {用电器};

  % Live conductor: supply -> breaker -> RCD -> load.
  \draw[live,->] (supply.east |- 0,1.15) -- (breaker.west);
  \draw[live,->] (breaker.east) -- (4.15,1.05);
  \draw[live,->] (4.15,1.05) -- (6.35,1.05);
  \draw[live,->] (6.35,1.05) -- (load.west |- 0,1.05);

  % Neutral conductor: load -> RCD -> supply.
  \draw[neutral,->] (load.west |- 0,0.25) -- (6.35,0.25);
  \draw[neutral,->] (6.35,0.25) -- (4.15,0.25);
  \draw[neutral,->] (4.15,0.25) -- (supply.east |- 0,0.25);

  \node[red!80!black,above] at (1.25,1.15) {火线 $L$};
  \node[blue!70!black,below] at (1.25,0.25) {零线 $N$};

  % Differential-current detector: both conductors pass through one ring.
  \draw[orange!85!black,thick] (5.25,0.65) ellipse (0.42 and 0.58);
  \node[font=\scriptsize,align=center] at (5.25,-0.63)
    {检测两线电流差：$I_\Delta=|I_L-I_N|$};

  % Metal enclosure and protective earth.
  \draw[thick,fill=white] (7.35,-1.30) rectangle (8.75,-0.68);
  \node at (8.05,-0.99) {金属外壳};
  \draw[thick] (load.south) -- (8.05,-0.68);
  \draw[earth,->] (8.05,-1.30) -- (8.05,-2.05);
  \draw[earth] (7.62,-2.05) -- (8.48,-2.05);
  \draw[earth] (7.78,-2.22) -- (8.32,-2.22);
  \draw[earth] (7.92,-2.38) -- (8.18,-2.38);
  \node[green!45!black,anchor=west] at (8.25,-1.72) {保护地线 $PE$};

  % Fault path.
  \draw[red!80!black,dashed,->] (7.55,0.22) -- (7.65,-0.68);
  \draw[green!45!black,dashed,->] (7.65,-0.68) -- (8.05,-1.30);
  \node[font=\scriptsize,align=center] at (2.55,2.05)
    {过载或短路\\自动断开};
  \draw[orange!80!black,->] (2.55,1.72) -- (2.55,1.52);
  \node[font=\scriptsize,align=left,anchor=west] at (8.95,0.55)
    {正常：$I_L=I_N$\\漏电：$I_\Delta\ne0$};
\end{tikzpicture}
