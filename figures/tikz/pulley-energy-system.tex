% TikZ 图：光滑桌面-定滑轮连接体的机械能守恒
% Day 09：机械能守恒与功能关系
\begin{tikzpicture}[scale=0.9,>=Latex]
  \coordinate (O) at (4.05,0);
  \def\r{0.32}

  % 墙角：定滑轮圆心固定在桌边墙角处
  \draw[thick] (-0.5,0) -- (O) -- (4.05,-2.85);
  \foreach \x in {-0.3,0.1,0.5,0.9,1.3,1.7,2.1,2.5,2.9,3.3,3.7} {
    \draw[gray] (\x,0) -- ++(-0.18,-0.18);
  }
  \foreach \y in {-0.15,-0.55,-0.95,-1.35,-1.75,-2.15,-2.55} {
    \draw[gray] (4.05,\y) -- ++(0.14,0.20);
  }

  \draw[thick] (1.0,0) rectangle (1.9,0.55);
  \node at (1.45,0.28) {$m_A$};
  \draw[->,thick] (1.9,0.28) -- (2.8,0.28) node[midway,above] {$v$};

  \draw[thick,fill=white] (O) circle (\r);
  \fill (O) circle (0.025);

  % 一根连续轻绳：m_A 端--定滑轮--m_B 端
  \draw[thick] (1.9,\r) -- ($(O)+(0,\r)$)
    arc[start angle=90,end angle=0,radius=\r]
    -- ($(O)+(\r,-1.25)$);

  \draw[thick] (4.08,-2.00) rectangle (4.66,-1.25);
  \node at (4.37,-1.63) {$m_B$};
  \draw[->,thick] (4.82,-1.25) -- (4.82,-2.05) node[midway,right] {$v$};
  \draw[->,thick,blue] (4.37,-2.00) -- (4.37,-2.62) node[midway,right] {$m_Bg$};
  \draw[dashed] (3.85,-2.70) -- (4.82,-2.70);
  \draw[<->] (3.62,-1.63) -- (3.62,-2.70) node[midway,left] {$h$};

  \node at (2.65,-3.18) {$m_Bgh=\frac12(m_A+m_B)v^2$};
\end{tikzpicture}
