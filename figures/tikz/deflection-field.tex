  \draw[thick] (0,1.5) -- (3,1.5);

  \filldraw[red] (-1,0) circle (2pt);

  \filldraw[red] (3.5,-1.5) circle (2pt);
  % 偏转量
  \draw[<->] (3.8,0) -- (3.8,-1.5);

  % 公式
