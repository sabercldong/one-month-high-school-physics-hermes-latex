\begin{tikzpicture}[scale=0.9,>=Latex,font=\small]
  \coordinate (P) at (3.0,-1.08);

  % 平行板
  \draw[thick] (0,1.5) -- (3,1.5);
  \node[above] at (1.5,1.5) {$+$};
  \draw[thick] (0,-1.5) -- (3,-1.5);
  \node[below] at (1.5,-1.5) {$-$};
  \foreach \x in {0.5,1.5,2.5} {
    \draw[->,blue] (\x,1.3) -- (\x,-1.3);
  }

  % 入射粒子与速度
  \filldraw[red] (-0.85,0) circle (2.2pt);
  \node[below left,red] at (-0.85,0) {$+q$};
  \draw[->,thick,red] (-0.70,0) -- (0.35,0);
  \node[above,red] at (-0.18,0) {$\vect v_0$};

  % 有场区轨迹与出口切向速度

    plot (\x,{-0.12*\x*\x});
  \filldraw[red] (P) circle (2.2pt);
  \draw[->,red,thick] (P) -- (3.82,-1.67);
  \node[red,above right] at (3.52,-1.43) {$\vect v$};

  % 出口偏转量
