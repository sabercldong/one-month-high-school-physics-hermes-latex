% TikZ 图：汽车水平转弯
% Day 11：圆周运动
\begin{tikzpicture}[scale=0.82,>=Latex,line cap=round,line join=round]
  \coordinate (C) at (0,-2.10);
  \coordinate (P) at (0,0);

  % 俯视弯道：两条同心圆弧
  \draw[thick] (C) ++(124:2.55) arc[start angle=124,end angle=56,radius=2.55];
  \draw[thick] (C) ++(128:1.65) arc[start angle=128,end angle=52,radius=1.65];
  \draw[dashed,gray] (C) -- (P);
  \fill[black] (C) circle (1.4pt);
  \node[right=4pt,font=\small] at (C) {圆心};

  % 汽车俯视图，车身沿切线方向
  \draw[thick,fill=gray!8] (-0.62,-0.30) rectangle (0.62,0.30);
  \node at (0,0) {车};

  % 速度沿切线，静摩擦力指向圆心
  \draw[->,thick,blue] (0.62,0) -- (1.55,0) node[right] {$v$};
  \draw[->,thick,red!80!black] (0,-0.32) -- (0,-1.35)
    node[midway,right=2pt] {$f_s$};
  \node[red!80!black,align=center,font=\small] at (-1.28,-0.85) {静摩擦力\\指向圆心};

\end{tikzpicture}
