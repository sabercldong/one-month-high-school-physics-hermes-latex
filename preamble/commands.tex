% 常用命令
\newcommand{\dd}{\mathrm{d}}
\newcommand{\ee}{\mathrm{e}}
\newcommand{\ii}{\mathrm{i}}
\newcommand{\vect}[1]{\boldsymbol{#1}}
% \unit is provided by siunitx package; we use \unsi as shorthand
\newcommand{\unsi}[1]{\,\mathrm{#1}}

% 每日讲义标题
\newcommand{\daytitle}[2]{%
  \chapter*{Day #1：#2}
  \addcontentsline{toc}{chapter}{Day #1：#2}
}

% 图注命令（在 box 内或非浮动环境使用，自动使用 figure 计数器）
\newcommand{\figcaption}[1]{%
  \refstepcounter{figure}%
  \textbf{图 \thefigure：}#1%
  \addcontentsline{lof}{figure}{\protect\numberline{\thefigure}#1}%
}
