\daytitle{20}{闭合电路欧姆定律与实验电路}

\begin{learningbox}{本节导读}
前面电路章节中，通常默认电源是理想的（电压恒定）。实际上电源有内阻，电动势和路端电压并不相等。本节学习闭合电路欧姆定律、路端电压随负载的变化、电源输出功率的特性，以及测量电源电动势和内阻的实验方法。
\end{learningbox}

\textbf{本节读题路线：} 先把电源等效成电动势 $E$ 与内阻 $r$ 串联，再区分路端电压、内电压和外电阻上的电压。动态电路题按“外电阻变化 $\to$ 总电流变化 $\to$ 内电压变化 $\to$ 路端电压变化”的顺序推，避免跳过中间量直接判断结论。

\textbf{符号提醒：} 本节通常用大写 $R$ 表示外电阻或负载电阻，小写 $r$ 表示电源内阻；比较输出功率时的条件 $R=r$ 指的是“外电阻等于内阻”，不是同一个电阻。

\section*{一、学习目标}

\begin{enumerate}
  \item 能说出电源电动势 $E$ 和内阻 $r$ 分别描述什么。
  \item 能从电源内外电压分配推出闭合电路欧姆定律 $I = \dfrac{E}{R+r}$。
  \item 能用 $U = E - Ir$ 分析路端电压随外电阻变化的过程。
  \item 能说明纯电阻负载下，电源输出功率 $P_{\text{出}} = I^2R$ 在 $R = r$ 时最大。
  \item 能从 $U$-$I$ 图像的截距和斜率说明伏安法测 $E$ 和 $r$ 的原理。
\end{enumerate}
\section*{二、初高中衔接}

初中物理中，电源常被近似看成恒定电压源（如“一节干电池 $1.5\ \mathrm{V}$”）。高中要进一步区分电动势和路端电压：空载时万用表读数常接近电动势；一旦接上负载，电流流过内阻，路端电压就会低于电动势。旧电池的内阻通常较大，带负载时电压下降更明显。

高中引入两个新概念：
\begin{itemize}
  \item \textbf{电动势 $E$：} 描述电源把其他形式的能转化为电能的本领（如一节干电池的电动势约为 $1.5\ \mathrm{V}$）。
  \item \textbf{内阻 $r$：} 电源内部电压损失的等效描述。内阻会随温度、荷电状态和老化程度变化，题目通常把给定工作状态下的 $r$ 视为定值。
\end{itemize}

闭合电路欧姆定律揭示了：$E$ = 外电路电压 + 内电路电压。

\textbf{科学发展视角：} 电池技术是新能源汽车、储能电站和移动设备的核心。内阻决定了电池大电流输出时的发热、效率和路端电压下降，所以“旧电池无法正常驱动负载”并非单纯经验判断，而是闭合电路欧姆定律的直接体现。

\section*{三、核心概念}

\begin{conceptbox}{电动势与闭合电路欧姆定律}

\textbf{电动势 $E$：} 电源非静电力把单位正电荷从负极搬到正极所做的功，反映电源把其他形式能转化为电能的本领。数值上，理想电压表测得的开路电压等于电动势。

\textbf{闭合电路欧姆定律：}
\[
\boxed{I = \frac{E}{R + r}}
\]

\textbf{它从哪里来：} 电源内部的非静电力把单位正电荷获得的能量记作 $E$。电荷沿闭合回路运动一周，这部分能量分别消耗在外电路和电源内阻上：
\[
E=U_{\text{外}}+U_{\text{内}}.
\]
若外电路为纯电阻 $R$，内阻为 $r$，同一闭合回路中电流相同，于是
\[
U_{\text{外}}=IR,\qquad U_{\text{内}}=Ir.
\]
代入得到
\[
E=IR+Ir=I(R+r),\qquad I=\frac{E}{R+r}.
\]
所以闭合电路欧姆定律不是孤立新增的公式，而是“电源提供的能量 = 外电路消耗 + 内阻消耗”的结果。

\begin{itemize}
  \item $E$：电源电动势（$\mathrm{V}$），由电源本身决定。
  \item $R$：外电路总电阻。
  \item $r$：电源内阻。
\end{itemize}

\end{conceptbox}

\begin{conceptbox}{路端电压与电压分配}

\textbf{路端电压（外电路电压）：}
\[
\boxed{U = IR = E - Ir}
\]

\begin{itemize}
  \item $U$ 是电源两端的实际电压（不等于 $E$）。
  \item $Ir$ 是内阻上的电压降（内电压）。
\end{itemize}

\end{conceptbox}

\input{figures/tikz/closed-circuit}

\figcaption{闭合电路：$E = IR + Ir = U + Ir$}\label{fig:closed-circuit}
\end{center}

\begin{conceptbox}{路端电压与外电阻的关系}

\textbf{以下变化规律均以电源的 $E$、$r$ 保持不变，电源处于放电状态为前提。}
\begin{itemize}
  \item 外电阻 $R \uparrow$ $\to$ 电流 $I \downarrow$ $\to$ 内压 $Ir \downarrow$ $\to$ 路端电压 $U \uparrow$。
  \item 外电阻 $R \downarrow$ $\to$ 电流 $I \uparrow$ $\to$ 内压 $Ir \uparrow$ $\to$ 路端电压 $U \downarrow$。
\end{itemize}

\textbf{两个极端情况：}
\begin{itemize}
  \item $R \to \infty$（开路）：$I = 0$，$U = E$（路端电压等于电动势）。
  \item $R \to 0$（短路）：$I_{\text{短}} = \dfrac{E}{r}$，$U = 0$。短路电流很大，会损坏电源。
\end{itemize}

\end{conceptbox}

\begin{conceptbox}{电源的输出功率}

\[
P_{\text{出}} = UI
\]
若外电路是纯电阻 $R$，则
\[
P_{\text{出}} = I^2R = \left(\frac{E}{R+r}\right)^2 R
\]

\textbf{最大输出功率条件：} 当 $E$、$r$ 不变且 $R>0$、$r>0$ 时，
\[
P_{\text{出}}
=\frac{E^2R}{(R+r)^2}
=\frac{E^2}{R+2r+\dfrac{r^2}{R}}.
\]
由基本不等式
\[
R+\frac{r^2}{R}\geq 2r
\]
可知分母的最小值为 $4r$，等号在 $R=r$ 时成立。因此
\[
\boxed{R=r},\qquad
P_{\max} = \frac{E^2}{4r}
\]

\begin{center}
\input{figures/tikz/power-output}

\figcaption{电源输出功率 $P_{\text{出}}$ 随外电阻 $R$ 的变化}\label{fig:power-output}
\end{center}

电源的效率：
\[
\eta = \frac{P_{\text{出}}}{P_{\text{总}}} = \frac{I^2R}{I^2(R+r)} = \frac{R}{R+r}
\]

$R$ 越大，效率越高。但当 $R\to\infty$ 时，$\eta\to 1$ 的同时 $I\to 0$、$P_{\text{出}}\to 0$。在最大输出功率时（$R = r$），效率仅为 $50\%$；因此“效率较高”和“输出功率较大”是两个不同的目标。

\end{conceptbox}

\begin{learningbox}{闭合电路的电压关系与功率关系}
闭合电路题先把“电源内部”和“外电路”分成两段来看：
\textbf{电压关系：} 电动势 $E$ 是电源提供的总能量对应的电压，路端电压 $U$ 是外电路得到的电压，内电压 $Ir$ 是电源内部消耗的电压；普通放电时 $E=U+Ir$。

\textbf{功率关系：} 电源总功率 $P_{\text{总}}=EI$，对外输出功率 $P_{\text{出}}=UI$，内阻发热功率 $P_{\text{内}}=I^2r$，三者满足 $EI=UI+I^2r$。

\textbf{动态分析：} 判断“电压变大或变小”时，先沿变化链条推理：外电路总电阻如何变，再由 $I=\dfrac{E}{R+r}$ 得到电流，最后回到 $U=E-Ir$ 和各支路电压。

\textbf{来源理解：} 闭合电路欧姆定律来自一圈电路的能量关系：电源每通过电荷 $q$ 提供能量 $qE$，外电路和内阻分别消耗 $qU$、$qIr$。因此普通放电时先写 $E=U+Ir$，若外电路等效为纯电阻 $R$，再得到 $I=\dfrac{E}{R+r}$。
\end{learningbox}

\subsection*{模型 2：电源伏安特性曲线}

$U = E - Ir$，$U$-$I$ 图像是一条斜率为 $-r$ 的直线。
\begin{itemize}
  \item 纵轴截距：$E$（电动势，$I=0$ 时）。
  \item 斜率绝对值：$r$（内阻）。
  \item 横轴截距：$I_{\text{短}}=\dfrac{E}{r}$（把直线外推到 $U=0$）。
\end{itemize}

\subsection*{模型 3：伏安法测 $E$ 和 $r$}

实验原理：改变外电阻 $R$，测多组 $U$、$I$，作 $U$-$I$ 图，由截距和斜率求 $E$ 和 $r$。图中先按理想电表处理：电压表内阻无限大、电流表内阻为零，此时电压表读路端电压，电流表读干路电流。

\begin{center}
\input{figures/tikz/source-measure-circuit}

\figcaption{伏安法测电源电动势和内阻：改变滑动变阻器，读取多组 $U$、$I$}\label{fig:source-measure-circuit}
\end{center}

\begin{center}
\input{figures/tikz/source-ui-data-fit}

\figcaption{多组实验数据作 $U$-$I$ 图：截距为 $E$，斜率为 $-r$}\label{fig:source-ui-data-fit}
\end{center}

若只给两组数据，也可以直接联立：
\[
r=\frac{U_1-U_2}{I_2-I_1},\qquad E=U_1+I_1r
\]

\textbf{实际仪表的误差判断：} 图中若电压表内阻为有限值 $R_V$，电流表读数记为 $I_A$，则电压表还分走电流 $I_V=U/R_V$。电源中的总电流是
\[
I_{\text{总}}=I_A+\frac{U}{R_V},
\]
而不是 $I_A$。代入 $U=E-rI_{\text{总}}$，整理得
\[
U=\frac{ER_V}{R_V+r}-\frac{rR_V}{R_V+r}I_A.
\]
若仍把纵截距和斜率绝对值直接当作电动势、内阻，就会得到
\[
E_{\text{测}}=\frac{ER_V}{R_V+r}<E,\qquad
r_{\text{测}}=\frac{rR_V}{R_V+r}<r.
\]
因此误差结论必须和具体接法对应；不能离开电路图孤立地背“偏大”或“偏小”。

\section*{六、例题精讲}

\begin{examplebox}{闭合电路欧姆定律}

\textbf{题意分析：} 电源电动势 $E = 12\ \mathrm{V}$，内阻 $r = 1\ \Omega$，外电阻 $R = 5\ \Omega$。求：
\begin{enumerate}
  \item 电路中的电流。
  \item 路端电压。
  \item 内电压。
\end{enumerate}

\textbf{物理模型：} 闭合电路欧姆定律。

\begin{center}
\input{figures/tikz/closed-circuit-example-basic}

\figcaption{例题电路：电源内阻与外电阻在同一闭合回路中}\label{fig:closed-circuit-example-basic}
\end{center}

\textbf{解题过程：}

\begin{enumerate}
  \item $I = \dfrac{E}{R+r} = \dfrac{12}{5+1} = 2\ \mathrm{A}$

  \item $U = IR = 2 \times 5 = 10\ \mathrm{V}$（或 $U = E - Ir = 12 - 2 \times 1 = 10\ \mathrm{V}$）

  \item $U_{\text{内}} = Ir = 2 \times 1 = 2\ \mathrm{V}$

  验证：$E = U + U_{\text{内}} = 10 + 2 = 12\ \mathrm{V}$ $\checkmark$
\end{enumerate}

\textbf{方法总结：} 在电源向外供电的普通闭合电路中，$E = U_{\text{外}} + U_{\text{内}}$。此时路端电压小于电动势，差值等于内阻上的电压降；若讨论给电池充电等反向过程，路端电压与电动势的关系需要重新按能量方向判断。

\end{examplebox}

\begin{examplebox}{路端电压的变化}

\textbf{题意分析：} 电源 $E = 6\ \mathrm{V}$，$r = 0.5\ \Omega$。外电阻从 $R_1 = 4.5\ \Omega$ 减小到 $R_2 = 1\ \Omega$，求电流和路端电压的变化。

\textbf{物理模型：} 闭合电路动态分析。

\begin{center}
\input{figures/tikz/terminal-voltage-change}

\figcaption{外电阻减小时，干路电流增大，内压增大，路端电压减小}\label{fig:terminal-voltage-change}
\end{center}

\textbf{解题过程：}

$R_1 = 4.5\ \Omega$：
\[
I_1 = \frac{6}{4.5 + 0.5} = \frac{6}{5} = 1.2\ \mathrm{A},\qquad
U_1 = 6 - 1.2 \times 0.5 = 5.4\ \mathrm{V}
\]

$R_2 = 1\ \Omega$：
\[
I_2 = \frac{6}{1 + 0.5} = \frac{6}{1.5} = 4\ \mathrm{A},\qquad
U_2 = 6 - 4 \times 0.5 = 4\ \mathrm{V}
\]

\textbf{方法总结：} 在 $E$、$r$ 不变时，动态电源题先由外电阻判断电流，再看内电压和路端电压；外电阻越小，电流越大，路端电压越小。

\end{examplebox}

\begin{examplebox}{电源最大输出功率}

\textbf{题意分析：} 电源 $E = 10\ \mathrm{V}$，$r = 2\ \Omega$。
\begin{enumerate}
  \item 外电阻为多大时输出功率最大？
  \item 最大输出功率是多少？
  \item 此时电源的效率是多少？
\end{enumerate}

\textbf{物理模型：} 电源输出功率。

\begin{center}
\input{figures/tikz/max-power-example}

\figcaption{本题输出功率曲线：最大点对应 $R=r=2\ \Omega$}\label{fig:max-power-example}
\end{center}

\textbf{解题过程：}

\begin{enumerate}
  \item $R = r = 2\ \Omega$ 时输出功率最大。

  \item $P_{\max} = \dfrac{E^2}{4r} = \dfrac{10^2}{4 \times 2} = \dfrac{100}{8} = 12.5\ \mathrm{W}$

  \item $\eta = \dfrac{R}{R+r} = \dfrac{2}{2+2} = 50\%$
\end{enumerate}

\textbf{方法总结：} 最大输出功率题先区分功率和效率；$R=r$ 时输出功率最大，但效率只有 $50\%$。

\end{examplebox}

  \item $R=2\ \Omega$ 时，$I_1=2.0\ \mathrm A$、$U_1=4.0\ \mathrm V$、

\end{enumerate}

\section*{八、竞赛入门补充}

\subsection*{1.\quad 两个负载的输出功率相等}

\begin{competitionbox}
\textbf{问题：} 同一电源分别接纯电阻 $R_1$、$R_2$，若两次输出功率相等，两个负载之间有什么关系？

由输出功率公式交叉相乘并整理：
\[
\begin{aligned}
\frac{E^2R_1}{(R_1+r)^2}
  &=\frac{E^2R_2}{(R_2+r)^2}\\
  &\Longrightarrow (R_1-R_2)(r^2-R_1R_2)=0.
\end{aligned}
\]
若 $R_1\ne R_2$，则 $\boxed{R_1R_2=r^2}$。这说明最大功率点 $R=r$
两侧可以出现两个输出功率相同的负载：一个小于 $r$，另一个大于 $r$。
二者输出功率虽然相同，但由 $\eta=R/(R+r)$ 可知，外阻较大的那一次效率更高。
\end{competitionbox}

\subsection*{2.\quad 含电容器的闭合电路}

\begin{competitionbox}
\textbf{含电容器的闭合电路}

在闭合电路中加入电容器，稳定时电容器支路相当于断路（没有电流通过），但电容器两端的电压等于并联部分的电压。

\begin{center}
\input{figures/tikz/capacitor-closed-circuit}

\figcaption{含电容器的闭合电路：稳定后电容器支路断路，$U_C=U_{R_2}$}\label{fig:capacitor-closed-circuit}
\end{center}

\textbf{例题：} 电源 $E = 10\ \mathrm{V}$，$r = 1\ \Omega$，$R_1 = 4\ \Omega$，$R_2 = 5\ \Omega$，电容器 $C = 10\ \mu\mathrm{F}$ 与 $R_2$ 并联。求电容器带电量。

\textbf{解：} 先求干路电流（电容器支路断路，不影响总电阻）：
\[
I = \frac{E}{R_1 + R_2 + r} = \frac{10}{4 + 5 + 1} = 1\ \mathrm{A}
\]
电容器两端电压等于 $R_2$ 两端电压：$U_C = IR_2 = 1 \times 5 = 5\ \mathrm{V}$

电量：$Q = CU_C = 10 \times 10^{-6} \times 5 = 5 \times 10^{-5}\ \mathrm{C}$

\textbf{竞赛提示：} 含电容器的直流电路问题中，稳定时电容器支路视为断路，电容器电压等于与之并联部分的电压。
\end{competitionbox}

\section*{九、易错点}

\begin{mistakebox}
\begin{enumerate}
  \item \textbf{混淆电动势和路端电压：} 先看电路是否闭合。电动势 $E$ 是电源本身的属性，路端电压 $U$ 是接入外电路后的实际电压；开路时 $U = E$，闭路有电流且内阻不为零时 $U = E-Ir < E$。

  \item \textbf{短路电流的计算忘记内阻：} 短路时先保留电源内阻，$R = 0$，$I = \dfrac{E}{r}$，不是 $\infty$。

  \item \textbf{动态分析中 $U$、$I$ 的变化方向判断错：} 先确认 $E$、$r$ 不变，再由 $I=\dfrac{E}{R+r}$ 判断总电流变化，最后由 $U=E-Ir$ 判断路端电压变化；例如 $R\uparrow$ 时，$I\downarrow$，内电压 $Ir\downarrow$，所以 $U\uparrow$。

  \item \textbf{最大功率的条件写反：} 先区分输出功率和效率；输出功率最大条件是 $R = r$，不是 $R$ 越大越好。

  \item \textbf{实验测 $E$、$r$ 的误差分析：} 先明确本题是否把电表视为理想。若非理想，先补画电压表内阻和电流表内阻，再判断电压表读的是哪一段电压、电流表读的是哪一路电流，最后才判断 $E$、$r$ 的系统误差方向。
\end{enumerate}
\end{mistakebox}

\section*{十、今日小结}

\begin{enumerate}
  \item 外电路可等效为纯电阻 $R$ 时，先画出内阻 $r$ 和外阻 $R$，再写闭合电路欧姆定律 $I = \dfrac{E}{R+r}$，$E = IR + Ir$。
  \item 在 $E$、$r$ 不变且电源放电时，判断路端电压变化要先看电流 $I$ 怎样变，再由 $U = E - Ir$ 判断；外电阻增大使 $I$ 减小、$U$ 增大。
  \item 电源输出功率 $P_{\text{出}} = I^2R$，讨论最大值前先确认只有 $R$ 改变、$E,r$ 不变；$R = r$ 时最大，$P_{\max} = \dfrac{E^2}{4r}$。
  \item 电源效率 $\eta = \dfrac{R}{R+r}$，要和最大输出功率分开判断；$R\to\infty$ 时效率趋近 $100\%$，但输出功率趋近于零。
  \item 伏安法 $U$-$I$ 图先写成 $U=E-rI$，再读纵截距 $=E$、斜率 $=-r$、横轴截距 $=E/r$；非理想电表的误差必须结合具体接法判断。
\end{enumerate}

\textbf{一分钟复盘：} 画一个含内阻电源的闭合电路，分别标 $E$、$Ir$、$U$ 和外电阻 $R$；再说出 $R$ 增大时 $I$、$U$、效率怎样变，并用 $R$ 与 $r$ 的大小关系判断输出功率的变化。
