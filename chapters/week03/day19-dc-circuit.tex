\daytitle{19}{恒定电流与基本电路}

\begin{learningbox}{本讲义大致内容}
从电荷的定向移动开始，理解电流、电阻、欧姆定律，再到串联和并联电路的分析方法。这些是电学实验和后续复杂电路分析的基础。
\end{learningbox}

\section*{一、学习目标}

\begin{enumerate}
  \item 理解电流 $I = \dfrac{Q}{t}$、电阻 $R = \dfrac{U}{I}$、欧姆定律 $I = \dfrac{U}{R}$。
  \item 掌握电阻定律 $R = \rho\dfrac{l}{S}$，理解电阻率 $\rho$ 的含义。
  \item 会分析串联和并联电路中 $I$、$U$、$R$ 的分配关系。
  \item 掌握电功 $W = IUt$ 和电功率 $P = IU$。
  \item 能进行简单的电路化简计算。
\end{enumerate}

\section*{二、初高中衔接}

初中已经学过这些基本知识——电流、电压、电阻、欧姆定律、串联并联。高中在以下方面深化：
\begin{enumerate}
  \item \textbf{电流的微观解释：} $I = nqSv$（$n$ 为载流子浓度）。
  \item \textbf{电阻定律：} $R = \rho \dfrac{l}{S}$——电阻与材料、长度、横截面积的定量关系。
  \item \textbf{电功率的精确理解：} $P = IU$ 对任何电路成立，在纯电阻电路中可改写为 $P = I^2R = \dfrac{U^2}{R}$。
  \item \textbf{电路化简和复杂计算：} 多电阻串并联的等效电阻计算。
\end{enumerate}

\section*{三、核心概念}

\begin{conceptbox}{电流与电阻}

\textbf{电流：}
\[
I = \frac{Q}{t}
\]
单位：安培（$\mathrm{A}$）。方向：正电荷定向移动的方向（与电子移动方向相反）。

\textbf{欧姆定律：}
\[
\boxed{I = \frac{U}{R}}
\]

\begin{itemize}
  \item 适用条件：金属导体和电解液（线性元件）。不适用于气体导电和半导体。
  \item $R$ 是导体本身的性质，与 $U$、$I$ 无关。
\end{itemize}

\textbf{电阻定律：}
\[
\boxed{R = \rho \frac{l}{S}}
\]

\begin{itemize}
  \item $\rho$：电阻率（$\Omega \cdot \mathrm{m}$），由材料本身决定。
  \item 金属的电阻率随温度升高而增大。
  \item 超导现象：温度低于临界温度时 $\rho = 0$。
\end{itemize}

\end{conceptbox}

\begin{conceptbox}{串联与并联电路}

\begin{center}
\input{figures/tikz/series-parallel}

\figcaption{串联（左）和并联（右）电路对比}\label{fig:series-parallel}
\end{center}

\textbf{串联电路：}
\begin{itemize}
  \item 电流处处相等：$I = I_1 = I_2$。
  \item 总电压等于各分电压之和：$U = U_1 + U_2$。
  \item 总电阻：$R = R_1 + R_2$。
  \item 电压分配：$\dfrac{U_1}{U_2} = \dfrac{R_1}{R_2}$（串联分压）。
\end{itemize}

\textbf{并联电路：}
\begin{itemize}
  \item 各支路电压相等：$U = U_1 = U_2$。
  \item 总电流等于各支路电流之和：$I = I_1 + I_2$。
  \item 总电阻：$\dfrac{1}{R} = \dfrac{1}{R_1} + \dfrac{1}{R_2}$。
  \item 电流分配：$\dfrac{I_1}{I_2} = \dfrac{R_2}{R_1}$（并联分流）。
\end{itemize}

\end{conceptbox}

\begin{conceptbox}{电功与电功率}

\textbf{电功：} 电流通过导体时做的功。
\[
W = IUt
\]

\textbf{焦耳定律（电热）：}
\[
Q = I^2 Rt
\]

\textbf{电功率：}
\[
\boxed{P = IU}
\]

\begin{itemize}
  \item 纯电阻电路（如电炉、灯泡）：$P = IU = I^2R = \dfrac{U^2}{R}$。
  \item 非纯电阻电路（如电动机）：$P = IU > I^2R$，多出的功率转化为机械能。
\end{itemize}

\end{conceptbox}

\section*{四、公式与适用条件}

\begin{center}
\begin{tabular}{lll}
\toprule
公式 & 物理意义 & 适用条件 \\
\midrule
$I = \dfrac{U}{R}$ & 欧姆定律 & 金属导体、电解液 \\
$R = \rho\dfrac{l}{S}$ & 电阻定律 & 温度不变 \\
$P = IU$ & 电功率 & 任何电路 \\
$Q = I^2Rt$ & 焦耳定律 & 任何电路 \\
串：$R = R_1 + R_2$ & 串联总电阻 & 串联 \\
并：$\dfrac{1}{R} = \dfrac{1}{R_1} + \dfrac{1}{R_2}$ & 并联总电阻 & 并联 \\
\bottomrule
\end{tabular}
\end{center}

\section*{五、典型模型}

\subsection*{模型 1：串联分压 / 并联分流}

串分压：电压与电阻成正比，大电阻分大电压。
并分流：电流与电阻成反比，大电阻分小电流。

\subsection*{模型 2：电路化简}

找等电位点 $\to$ 判断串并联关系 $\to$ 逐级求等效电阻。

\subsection*{模型 3：非纯电阻电路}

电动机：$P_{\text{电}} = UI$，$P_{\text{热}} = I^2R$，$P_{\text{机}} = UI - I^2R$。

\section*{六、例题精讲}

\begin{examplebox}{串并联电路的计算}

\textbf{题意分析：} $R_1 = 6\ \Omega$，$R_2 = 12\ \Omega$ 并联后与 $R_3 = 4\ \Omega$ 串联，接在 $U = 12\ \mathrm{V}$ 的电源上。求：
\begin{enumerate}
  \item 电路总电阻。
  \item 干路电流。
  \item $R_1$、$R_2$、$R_3$ 各自的电压和电流。
\end{enumerate}

\textbf{物理模型：} 混联电路。

\textbf{解题过程：}

\begin{enumerate}
  \item $R_{12} = \dfrac{R_1 R_2}{R_1 + R_2} = \dfrac{6 \times 12}{6 + 12} = 4\ \Omega$
  
  $R_{\text{总}} = R_{12} + R_3 = 4 + 4 = 8\ \Omega$
  
  \item $I_{\text{总}} = \dfrac{U}{R_{\text{总}}} = \dfrac{12}{8} = 1.5\ \mathrm{A}$
  
  \item $U_3 = I_{\text{总}} R_3 = 1.5 \times 4 = 6\ \mathrm{V}$
  
  $U_{12} = U - U_3 = 12 - 6 = 6\ \mathrm{V}$（并联部分电压）
  
  $I_1 = \dfrac{U_{12}}{R_1} = \dfrac{6}{6} = 1\ \mathrm{A}$，$I_2 = \dfrac{U_{12}}{R_2} = \dfrac{6}{12} = 0.5\ \mathrm{A}$
\end{enumerate}

\textbf{方法总结：} 混联电路的解题顺序：先合并并联部分 $\to$ 求总电阻 $\to$ 求总电流 $\to$ 逐级反推各支路。

\end{examplebox}

\begin{examplebox}{电功率计算}

\textbf{题意分析：} 标有"$220\ \mathrm{V}$、$100\ \mathrm{W}$"的灯泡：
\begin{enumerate}
  \item 正常发光时电流和电阻是多少？
  \item 接在 $110\ \mathrm{V}$ 电源上时实际功率是多少？（假设电阻不变）
\end{enumerate}

\textbf{物理模型：} 纯电阻电路。

\textbf{解题过程：}

\begin{enumerate}
  \item $I = \dfrac{P}{U} = \dfrac{100}{220} \approx 0.455\ \mathrm{A}$
  
  $R = \dfrac{U^2}{P} = \dfrac{220^2}{100} = 484\ \Omega$
  
  \item $P_{\text{实}} = \dfrac{U_{\text{实}}^2}{R} = \dfrac{110^2}{484} = \dfrac{12100}{484} = 25\ \mathrm{W}$
\end{enumerate}

\textbf{方法总结：} 灯泡的电阻通常认为是定值。电压减半，功率减为 $\dfrac14$。

\end{examplebox}

\begin{examplebox}{非纯电阻电路}

\textbf{题意分析：} 一台电动机的线圈电阻 $R = 1\ \Omega$，接在 $U = 220\ \mathrm{V}$ 电源上工作时电流 $I = 10\ \mathrm{A}$。求：
\begin{enumerate}
  \item 输入电功率。
  \item 发热功率。
  \item 输出机械功率和效率。
\end{enumerate}

\textbf{物理模型：} 非纯电阻电路，$P = UI \neq I^2R$。

\textbf{解题过程：}

\begin{enumerate}
  \item $P_{\text{电}} = UI = 220 \times 10 = 2200\ \mathrm{W}$
  
  \item $P_{\text{热}} = I^2 R = 10^2 \times 1 = 100\ \mathrm{W}$
  
  \item $P_{\text{机}} = P_{\text{电}} - P_{\text{热}} = 2200 - 100 = 2100\ \mathrm{W}$
  
  效率 $\eta = \dfrac{P_{\text{机}}}{P_{\text{电}}} = \dfrac{2100}{2200} \approx 95.5\%$
\end{enumerate}

\textbf{方法总结：} 非纯电阻电路不能直接用 $P = I^2R$ 或 $P = U^2/R$ 算总功率，必须用 $P = UI$。发热功率一定用 $P = I^2R$。

\end{examplebox}

\section*{八、易错点}

\begin{mistakebox}
\begin{enumerate}
  \item \textbf{混联电路中串并联关系判断错：} 电流只有一条路径 $\to$ 串联；电流分叉 $\to$ 并联。找等电位点法是判断串并联的可靠方法。
  
  \item \textbf{非纯电阻电路用错公式：} 只有纯电阻才能用 $P = I^2R = U^2/R$ 算总功率。有电动机的电路中 $P = UI \neq I^2R$。
  
  \item \textbf{并联总电阻小于任何一个分电阻：} $n$ 个相同的电阻 $R$ 并联，总电阻为 $\dfrac{R}{n}$。
  
  \item \textbf{串联分压时大电阻分大电压：} 不是大电阻分大电流（串联电流相等，不存在分电流）。
  
  \item \textbf{误以为焦耳定律 $Q = I^2Rt$ 只适用于纯电阻电路：} 它适用于任何电路（电流的热效应总是存在的），但在非纯电阻电路中 $Q \neq IUt$。
\end{enumerate}
\end{mistakebox}

\section*{九、竞赛入门补充}

\begin{competitionbox}
\textbf{电路的化简技巧——对称法}

如果电路具有对称性，可以利用对称性简化计算。

\textbf{例题：} 一个正方体框架每条棱的电阻为 $R$，求体对角线两个顶点间的等效电阻。

\textbf{解：} 利用对称性，将与顶点相连的三条棱上的电流相同，将该顶点与对面顶点等电位的三个点合并，可将电路化简为一个简单的串并联组合。最终结果为 $R_{\text{总}} = \dfrac{5}{6}R$。

\textbf{竞赛小贴士：} 遇到复杂网络电阻问题，先观察对称性。等电位点可以合并、电流对称的支路可以等分，这些技巧能大幅简化计算。
\end{competitionbox}

\section*{十、今日小结}

\begin{enumerate}
  \item 欧姆定律 $I = \dfrac{U}{R}$，电阻定律 $R = \rho\dfrac{l}{S}$。
  \item 串联：$I$ 相等，$U$ 分压，$R = R_1 + R_2$。
  \item 并联：$U$ 相等，$I$ 分流，$\dfrac{1}{R} = \dfrac{1}{R_1} + \dfrac{1}{R_2}$。
  \item 电功率 $P = IU$，焦耳定律 $Q = I^2Rt$。
  \item 非纯电阻电路中 $P_{\text{电}} = UI$，$P_{\text{热}} = I^2R$，两者不等。
\end{enumerate}
